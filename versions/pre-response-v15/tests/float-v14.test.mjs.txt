import test from 'node:test';
import assert from 'node:assert/strict';
import {PalmPhysics,meshArea} from '../physics-v14.mjs';
import {once} from 'node:events';
import {createHarveyServer} from '../connection-v1/server-v1.mjs';
const seeded=()=>{let s=123;return()=>((s=(Math.imul(s,1664525)+1013904223)>>>0)/4294967296);};
function translate(b,x,y,vx=0,vy=0){const dx=x-b.x,dy=y-b.y;for(const p of b.particles){p.x+=dx;p.y+=dy;p.vx=vx;p.vy=vy;}}
function distortion(b){const r=b.nodes.map(p=>Math.hypot(p.x-b.x,p.y-b.y));return Math.max(...r)-Math.min(...r);}
test('v14 every blob keeps floating after four minutes without interaction',()=>{
 const p=new PalmPhysics({random:seeded()}),travel=Array(8).fill(0);let previous;
 for(let f=0;f<240*120;f++){
  p.advance(1/120);
  if(f===210*120)previous=p.blobs.map(b=>({x:b.x,y:b.y}));
  if(f>210*120)p.blobs.forEach((b,i)=>{travel[i]+=Math.hypot(b.x-previous[i].x,b.y-previous[i].y);previous[i]={x:b.x,y:b.y};});
 }
 for(const d of travel)assert.ok(d>150&&d<1200,`late idle travel ${d}`);
 assert.ok(p.stats.minAreaRatio>.98);assert.ok(p.stats.maxEdgeStretch<1.03);
});
test('v14 uniform drift moves a stationary mesh without gravity or generated deformation',()=>{
 const a=new PalmPhysics({count:1,random:()=>0}),b=new PalmPhysics({count:1,random:()=>0});
 for(const p of [a,b])translate(p.blobs[0],240,240);
 for(let f=0;f<120*3;f++){a.advance(1/120);b.advance(1/120,{}, {hasTilt:true,tiltY:9.81});}
 assert.ok(a.blobs[0].x>250);assert.ok(Math.abs(a.blobs[0].y-240)<1e-5);
 assert.ok(distortion(a.blobs[0])<.001);assert.equal(a.blobs[0].x,b.blobs[0].x);assert.equal(a.blobs[0].y,b.blobs[0].y);
});
test('v14 local deformation remains after release, relaxes gradually, and keeps travelling',()=>{
 const p=new PalmPhysics({count:1,random:()=>.5}),b=p.blobs[0];translate(b,240,240);p.advance(1/120);
 p.setGrab(b.nodes[0].x,b.nodes[0].y);p.setGrab(335,210);
 for(let f=0;f<180;f++)p.advance(1/120,{repulsion:.5,deformation:1,held:true});
 p.releaseGrab();const x=b.x,y=b.y,start=distortion(b);let atTwo,atFour,travelAtTwo;
 for(let f=0;f<12*120;f++){
  p.advance(1/120,{repulsion:0,deformation:0,held:false});
  if(f===239){atTwo=distortion(b);travelAtTwo=Math.hypot(b.x-x,b.y-y);}
  if(f===479)atFour=distortion(b);
 }
 assert.ok(start>3);assert.ok(atTwo>1.5);assert.ok(travelAtTwo>20);assert.ok(atFour<atTwo);assert.ok(distortion(b)<.5);
 assert.ok(meshArea(b)/b.restArea>.95);
});
test('v14 only physically strained links acquire a deformation memory',()=>{
 const p=new PalmPhysics({count:2,random:()=>0});translate(p.blobs[0],150,240);translate(p.blobs[1],340,240);
 p.advance(1/120);const b=p.blobs[0];p.setGrab(b.nodes[0].x,b.nodes[0].y);p.setGrab(185,195);
 for(let f=0;f<100;f++)p.advance(1/120,{deformation:1,repulsion:.5,held:true});
 assert.ok(Math.max(...b.links.map(l=>Math.abs(l.memory)))>.5);
 assert.ok(Math.max(...p.blobs[1].links.map(l=>Math.abs(l.memory)))<.01);
});
test('v14 six-minute interaction and recovery stress preserves finite, contained separate membranes',()=>{
 const p=new PalmPhysics({random:seeded()});
 for(let f=0;f<360*120;f++){
  const t=f/120,phase=Math.floor(t/8)%3;
  if(phase===1){if(!p.grab){const n=p.blobs[0].nodes[0];p.setGrab(n.x,n.y);}p.setGrab(240+Math.cos(t)*175,240+Math.sin(t)*175);}else p.releaseGrab();
  p.advance(1/120,{repulsion:phase===0?0:.8,deformation:phase===0?0:.9,held:phase!==0},phase===2?{x:Math.sin(t*12)*22,y:Math.cos(t*9)*18,omega:Math.sin(t)*2}:{});
  if(f%120)continue;
  for(const b of p.blobs){const ratio=meshArea(b)/b.restArea;assert.ok(ratio>.6&&ratio<1.4,`area ratio ${ratio}`);
   for(const n of b.nodes){assert.ok([n.x,n.y,n.vx,n.vy].every(Number.isFinite));assert.ok(Math.hypot(n.x-240,n.y-240)<=238.00001);}
  }
 }
 assert.equal(p.blobs.length,8);console.log('v14 stress',JSON.stringify(p.stats));
});
test('v14 floating and recovery are consistent at 30/60/120 drawing fps',()=>{
 let reference;
 for(const fps of [30,60,120]){const p=new PalmPhysics({random:seeded()});
  for(let i=0;i<fps*3;i++)p.advance(1/fps,{repulsion:.5,deformation:.8,held:true},{x:2,y:-1});
  const points=p.blobs.flatMap(b=>b.nodes.map(n=>({x:n.x,y:n.y})));if(!reference)reference=points;else points.forEach((n,i)=>assert.ok(Math.hypot(n.x-reference[i].x,n.y-reference[i].y)<1e-7));
 }
});

 const {server}=createHarveyServer();server.listen(0,'127.0.0.1');await once(server,'listening');t.after(()=>new Promise(resolve=>server.close(resolve)));
 assert.equal((await fetch(`http://127.0.0.1:${server.address().port}/portable/physics-v14.mjs`)).status,200);
 const app=await(await fetch(`http://127.0.0.1:${server.address().port}/portable/app.js`)).text();assert.match(app,/physics-v14/);assert.match(app,/held:interaction.held/);
});
